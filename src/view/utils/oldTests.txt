
// All tests use L=1, EI=1, EA=1000 (axial stiffness is irrelevant here),
// horizontal element (ni at origin, nj at (1,0)), zero node angles.
// DOF layout: [u_i, v_i, θ_i, u_j, v_j, θ_j] → indices 0–5.

function makeNodes(L = 1, angleDeg = 0): [StructuralNode, StructuralNode] {
    const a = (angleDeg * Math.PI) / 180;
    const ni: StructuralNode = {
        id: 1, x: 0, z: 0, mass: 0,
        restraint: { u: false, v: false, theta: false },
        angle: 0, dofs: [0, 1, 2],
    };
    const nj: StructuralNode = {
        id: 2, x: L * Math.cos(a), z: L * Math.sin(a), mass: 0,
        restraint: { u: false, v: false, theta: false },
        angle: 0, dofs: [3, 4, 5],
    };
    return [ni, nj];
}

function makeElement(
    ei: number,
    releases_i_theta = false,
    releases_j_theta = false,
): StructuralElement {
    return {
        id: 1, node_i: 1, node_j: 2,
        ea: 1000, ei, c: 0,
        dofs: [0, 1, 2, 3, 4, 5],
        releases_i: { u: false, v: false, theta: releases_i_theta },
        releases_j: { u: false, v: false, theta: releases_j_theta },
    };
}

// Build a getDof callback from an array of 6 values [u_i, v_i, θ_i, u_j, v_j, θ_j]
function dofs(values: number[]) {
    return (idx: number) => values[idx];
}

const ZERO = dofs([0, 0, 0, 0, 0, 0]);
const TOL  = 1e-10;

// ──────────────────────────────────────────────────────────────────────────────
// Hermite shape functions: pure displacement / rotation cases, no distributed load
// ──────────────────────────────────────────────────────────────────────────────

describe('Hermite interpolation (no load)', () => {
    test('zero DOFs → zero displacement everywhere', () => {
        const [ni, nj] = makeNodes();
        const el = makeElement(1);
        for (const xi of [0, 0.25, 0.5, 0.75, 1]) {
            const { u, v } = elementDisplacementAt(el, ni, nj, ZERO, xi);
            expect(Math.abs(u)).toBeLessThan(TOL);
            expect(Math.abs(v)).toBeLessThan(TOL);
        }
    });

    test('unit axial displacement at j → linear axial interpolation', () => {
        const [ni, nj] = makeNodes();
        const el = makeElement(1);
        const getDof = dofs([0, 0, 0, 1, 0, 0]);
        for (const xi of [0, 0.25, 0.5, 0.75, 1]) {
            const { u, v } = elementDisplacementAt(el, ni, nj, getDof, xi);
            expect(Math.abs(u - xi)).toBeLessThan(TOL);
            expect(Math.abs(v)).toBeLessThan(TOL);
        }
    });

    test('unit transverse settlement at j → cubic Hermite shape functions', () => {
        // v_j = 1, all other DOFs = 0  →  v(ξ) = N3 = 3ξ² - 2ξ³
        const [ni, nj] = makeNodes();
        const el = makeElement(1);
        const getDof = dofs([0, 0, 0, 0, 1, 0]);
        for (const xi of [0, 0.25, 0.5, 0.75, 1]) {
            const N3 = 3 * xi * xi - 2 * xi * xi * xi;
            const { v } = elementDisplacementAt(el, ni, nj, getDof, xi);
            expect(Math.abs(v - N3)).toBeLessThan(TOL);
        }
    });

    test('boundary values satisfy interpolation exactly', () => {
        const [ni, nj] = makeNodes();
        const el = makeElement(1);
        const getDof = dofs([0, 0.3, 0.1, 0, -0.2, -0.05]);
        const { u: u0, v: v0 } = elementDisplacementAt(el, ni, nj, getDof, 0);
        const { u: u1, v: v1 } = elementDisplacementAt(el, ni, nj, getDof, 1);
        expect(Math.abs(v0 - 0.3)).toBeLessThan(TOL);
        expect(Math.abs(v1 - (-0.2))).toBeLessThan(TOL);
        expect(Math.abs(u0)).toBeLessThan(TOL);
        expect(Math.abs(u1)).toBeLessThan(TOL);
    });
});

// ──────────────────────────────────────────────────────────────────────────────
// Particular solution: fixed-fixed beam (no releases, zero nodal DOFs)
// Analytical solutions are for L=1, EI=1.
// ──────────────────────────────────────────────────────────────────────────────

describe('Particular solution – fixed-fixed beam (L=1, EI=1)', () => {
    test('uniform load q=1 → v(0.5) = −1/384', () => {
        // Analytical: v_mid = qL⁴/(384EI) downward = −1/384
        const [ni, nj] = makeNodes();
        const el = makeElement(1);
        const { v } = elementDisplacementAt(el, ni, nj, ZERO, 0.5, { qi: 1, qj: 1 });
        expect(v).toBeCloseTo(-1 / 384, 12);
    });

    test('triangular load qi=0, qj=1 → v(0.5) = −1/768', () => {
        // v_p = −dq·(ξ⁵/120 − ξ³/40 + ξ²/60) at ξ=0.5, dq=1
        // = −(1/3840 − 12/3840 + 16/3840) = −5/3840 = −1/768
        const [ni, nj] = makeNodes();
        const el = makeElement(1);
        const { v } = elementDisplacementAt(el, ni, nj, ZERO, 0.5, { qi: 0, qj: 1 });
        expect(v).toBeCloseTo(-1 / 768, 12);
    });

    test('zero load → zero particular-solution contribution', () => {
        const [ni, nj] = makeNodes();
        const el = makeElement(1);
        for (const xi of [0.1, 0.3, 0.5, 0.7, 0.9]) {
            const { v } = elementDisplacementAt(el, ni, nj, ZERO, xi, { qi: 0, qj: 0 });
            expect(Math.abs(v)).toBeLessThan(TOL);
        }
    });

    test('boundary displacement is zero regardless of load', () => {
        // A fixed-fixed beam with nodal DOFs=0 must have v=0 at both ends
        const [ni, nj] = makeNodes();
        const el = makeElement(1);
        const load = { qi: 3, qj: 7 };
        const { v: v0 } = elementDisplacementAt(el, ni, nj, ZERO, 0, load);
        const { v: v1 } = elementDisplacementAt(el, ni, nj, ZERO, 1, load);
        expect(Math.abs(v0)).toBeLessThan(TOL);
        expect(Math.abs(v1)).toBeLessThan(TOL);
    });
});

// ──────────────────────────────────────────────────────────────────────────────
// Moment releases: back-calculation of end slopes
// ──────────────────────────────────────────────────────────────────────────────

describe('Moment releases – back-calculated end slopes (L=1, EI=1)', () => {
    test('simply supported (both ends released), uniform load → v(0.5) = −5/384', () => {
        // Analytical: 5qL⁴/(384EI) downward = −5/384
        const [ni, nj] = makeNodes();
        const el = makeElement(1, true, true);
        const { v } = elementDisplacementAt(el, ni, nj, ZERO, 0.5, { qi: 1, qj: 1 });
        expect(v).toBeCloseTo(-5 / 384, 12);
    });

    test('simply supported, uniform load → end slopes = ±qL³/(24EI) = ±1/24', () => {
        // θ(0)  = L·v′(0) = L·d/dξ(N2*L*θᵢ + N4*L*θⱼ)|ξ=0 = L*L*θᵢ
        // Since slope v′ at ξ=0 is θᵢ (in radians), the end angle is θᵢ.
        // We check that v′ at ξ=0 satisfies slope = −1/24 (clockwise at left pin).
        const [ni, nj] = makeNodes();
        const el = makeElement(1, true, true);
        const dxi = 1e-6;
        const { v: v0  } = elementDisplacementAt(el, ni, nj, ZERO, 0,       { qi: 1, qj: 1 });
        const { v: vdx } = elementDisplacementAt(el, ni, nj, ZERO, dxi, { qi: 1, qj: 1 });
        const slopeLeft = (vdx - v0) / dxi;
        expect(slopeLeft).toBeCloseTo(-1 / 24, 5);   // L=1, so dv/dξ = dv/dx
    });

    test('simply supported, triangular load qi=0, qj=1 → v(0.5) correct', () => {
        // Analytical: for simply-supported beam with triangular load (0→q over [0,L]):
        // At midspan, deflection = 5qL⁴/(768EI) downward = −5/768 (for L=EI=q=1)
        // Derived from superposition: v = v_hom(θ_i, θ_j) + v_p
        //   θ_i from M=0 at left:  θ_i = Mj_fem*L/(6EI) - Mi_fem*L/(3EI)
        //     Mi_fem = (3*0+2*1)/60 = 2/60 = 1/30
        //     Mj_fem = -(2*0+3*1)/60 = -3/60 = -1/20
        //     θ_i = (-1/20)/(6) - (1/30)/(3) = -1/120 - 1/90 = -3/360 - 4/360 = -7/360
        //   θ_j from M=0 at right: θ_j = Mi_fem*L/(6EI) - Mj_fem*L/(3EI)
        //     θ_j = (1/30)/(6) - (-1/20)/(3) = 1/180 + 1/60 = 1/180 + 3/180 = 4/180 = 1/45
        //   v_hom(0.5) = N2*θ_i + N4*θ_j = 0.125*(-7/360) + (-0.125)*(1/45)
        //              = -7/2880 - 1/360 = -7/2880 - 8/2880 = -15/2880 = -1/192
        //   v_p(0.5)   = -1/768  (from the triangular particular solution test above)
        //   v_total    = -1/192 - 1/768 = -4/768 - 1/768 = -5/768
        const [ni, nj] = makeNodes();
        const el = makeElement(1, true, true);
        const { v } = elementDisplacementAt(el, ni, nj, ZERO, 0.5, { qi: 0, qj: 1 });
        expect(v).toBeCloseTo(-5 / 768, 12);
    });

    test('fixed-pinned (release at i), uniform load → v(0.5) = −1/192', () => {
        // Analytical: 1/192 downward = −1/192 (≈ 0.005208 qL⁴/EI)
        // The fixed end (j) has th_j=0; back-calc gives θ_i = −1/48.
        // v_hom(0.5) = N2*L*θ_i = 0.125*(−1/48) = −1/384
        // v_p(0.5)   = −1/384
        // v_total    = −2/384 = −1/192
        const [ni, nj] = makeNodes();
        const el = makeElement(1, true, false);
        const { v } = elementDisplacementAt(el, ni, nj, ZERO, 0.5, { qi: 1, qj: 1 });
        expect(v).toBeCloseTo(-1 / 192, 12);
    });

    test('fixed-pinned (release at j), uniform load → v(0.5) = −1/192 by symmetry', () => {
        // Mirror of the above: release at j, fixed at i (th_i=0).
        // θ_j back-calc = +1/48; same magnitude gives same midspan deflection.
        const [ni, nj] = makeNodes();
        const el = makeElement(1, false, true);
        const { v } = elementDisplacementAt(el, ni, nj, ZERO, 0.5, { qi: 1, qj: 1 });
        expect(v).toBeCloseTo(-1 / 192, 12);
    });
});

// ──────────────────────────────────────────────────────────────────────────────
// Inclined element: verify DOF rotation (local u/v ← global DOFs)
// ──────────────────────────────────────────────────────────────────────────────

describe('Inclined element – DOF rotation', () => {
    test('vertical element (90°): global v-displacement maps to local axial u', () => {
        // thetaE = π/2 → local u = global v, local v = −global u
        const [ni, nj] = makeNodes(1, 90);
        const el = makeElement(1);
        // Apply global v=δ at node i only; expect u_i = δ, v_i = 0
        const getDof = dofs([0, 0.5, 0, 0, 0, 0]);
        const { u, v } = elementDisplacementAt(el, ni, nj, getDof, 0);
        expect(u).toBeCloseTo(0.5, 12);
        expect(v).toBeCloseTo(0,   12);
    });

    test('vertical element (90°): global u-displacement maps to local transverse v', () => {
        // u_i (global) → local v_i = −u_i
        const [ni, nj] = makeNodes(1, 90);
        const el = makeElement(1);
        const getDof = dofs([0.5, 0, 0, 0, 0, 0]);
        const { u, v } = elementDisplacementAt(el, ni, nj, getDof, 0);
        expect(u).toBeCloseTo(0,    12);
        expect(v).toBeCloseTo(-0.5, 12);
    });

    test('45° element: unit global u at i → u_i = 1/√2, v_i = −1/√2', () => {
        const [ni, nj] = makeNodes(1, 45);
        const el = makeElement(1);
        const getDof = dofs([1, 0, 0, 0, 0, 0]);
        const { u, v } = elementDisplacementAt(el, ni, nj, getDof, 0);
        expect(u).toBeCloseTo( 1 / Math.SQRT2, 12);
        expect(v).toBeCloseTo(-1 / Math.SQRT2, 12);
    });
});
